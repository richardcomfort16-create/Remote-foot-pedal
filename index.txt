function send(cmd) {
    fetch("/cmd?button=" + cmd)
        .then(r => console.log("Sent:", cmd))
        .catch(e => console.log("Error:", e));
}

// Colour override for ALL fixtures
function sendColourAll() {
    let c = document.getElementById("allColour").value;
    send("ALLCOLOR_" + c.substring(1));
}

// PAR-only colour
function sendColourPAR() {
    let c = document.getElementById("parColour").value;
    send("PARCOLOR_" + c.substring(1));
}

// Moving Head-only colour
function sendColourMH() {

}

// Joystick control
let stick, joy;
window.onload = () => {
    joy = document.getElementById("joystick");
    stick = document.getElementById("stick");

    joy.addEventListener("touchmove", handleJoy);
    joy.addEventListener("touchend", resetJoy);
};

function handleJoy(ev) {
    ev.preventDefault();
    let rect = joy.getBoundingClientRect();
    let x = ev.touches[0].clientX - rect.left - 100;
    let y = ev.touches[0].clientY - rect.top - 100;

    let limit = 70;
    let dx = Math.max(-limit, Math.min(limit, x));
    let dy = Math.max(-limit, Math.min(limit, y));

    stick.style.left = (dx + 100 - 30) + "px";
    stick.style.top = (dy + 100 - 30) + "px";

    let pan = Math.round((dx / limit) * 100);
    let tilt = Math.round((dy / limit) * 100);

    send("PT_" + pan + "_" + tilt);
}

function resetJoy() {
    stick.style.left = "70px";
    stick.style.top = "70px";
    send("PT_0_0");
}
